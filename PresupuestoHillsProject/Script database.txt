CREATE DATABASE PresupuestoDB;
USE PresupuestoDB;

CREATE TABLE Rol (
    id INT(1) PRIMARY KEY,
    descRol VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE Usuario (
    id INT(15) PRIMARY KEY,
    nDocId VARCHAR(50) NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(25) NOT NULL,
    rol INT NOT NULL,
    FOREIGN KEY (rol) REFERENCES Rol(id)
);

CREATE TABLE TipoPartida (
    id INT(1) PRIMARY KEY,
    descTipo VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE EstadoPresupuesto (
    id INT(1) PRIMARY KEY,
    descEstado VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE EstadoAprobacion (
    id INT(1) PRIMARY KEY,
    descEstado VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE CategoriaPresupuesto (
    id INT(3) PRIMARY KEY,
    nombreCategoria VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE PresupuestoGeneral (
    id INT(15) PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT,
    montoTotal DECIMAL(15,2) NOT NULL,
    estado_id INT NOT NULL,
    categoria_id INT NOT NULL,
    fechaCreacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (estado_id) REFERENCES EstadoPresupuesto(id),
    FOREIGN KEY (categoria_id) REFERENCES CategoriaPresupuesto(id)
);

CREATE TABLE PartidaPresupuestaria (
    id INT(10) PRIMARY KEY,
    categoria_id INT NOT NULL,
    descripcion TEXT NOT NULL,
    monto DECIMAL(15,2) NOT NULL,
    tipo_id INT NOT NULL,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (categoria_id) REFERENCES CategoriaPresupuesto(id),
    FOREIGN KEY (tipo_id) REFERENCES TipoPartida(id)
);

CREATE TABLE EjecucionPresupuestaria (
    id INT(10) PRIMARY KEY,
    partida_id INT NOT NULL,
    montoReal DECIMAL(15,2) NOT NULL,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (partida_id) REFERENCES PartidaPresupuestaria(id)
);

CREATE TABLE Aprobacion (
    id INT(20) PRIMARY KEY,
    presupuesto_id INT NOT NULL,
    usuario_id INT NOT NULL,
    estado_id INT NOT NULL,
    comentarios TEXT,
    fechaAprobacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (presupuesto_id) REFERENCES PresupuestoGeneral(id),
    FOREIGN KEY (usuario_id) REFERENCES Usuario(id),
    FOREIGN KEY (estado_id) REFERENCES EstadoAprobacion(id)
);
